598	#!/bin/bash
599	# CCR SessionStart hook: re-assert the git committer identity so that
600	# commits SSH-signed by CCR verify on GitHub.
601	#
602	# CCR's signing key is registered to noreply@anthropic.com. If a commit's
603	# committer email is anything else, GitHub reports verification.reason =
604	# unknown_key and shows the commit as "Unverified" even though the signature
605	# itself is valid. The base CCR image's /home/claude/.gitconfig sets
606	# commit.gpgsign=true but historically did not pin user.email, and past
607	# settings-sync behavior (tengu_strap_foyer) could pull in a user's own
608	# SessionStart hook that overwrote the identity. Re-asserting on every
609	# session start closes both gaps, including for checkpointed containers that
610	# still carry stale global git config.
611	#
612	# This sets both author and committer; CCR does not currently preserve a
613	# distinct human author identity in 1p sessions, and the Stop hook's
614	# remediation instructions and the byoc envtype already use these exact
615	# values. Exit 0 even on failure so a broken git install can't wedge session
616	# startup — the Stop hook will still catch any resulting unverifiable commit.
617	
618	git config --global user.email noreply@anthropic.com 2>/dev/null
619	git config --global user.name Claude 2>/dev/null
620	
621	# When the session has a user account, install a git commit-msg hook that
622	# appends a Co-authored-by trailer carrying the owner's address — the write
623	# side of the commit_identity_strict_committer rule in
624	# commitsign.MatchesAccountStrict. core.hooksPath replaces the repo-local
625	# hooks dir for every hook name, so every other name is a passthrough that
626	# chain-calls $GIT_DIR/hooks/<name> to keep repo-local hooks working.
627	#
628	# When CCR_SESSION_ACCOUNT_EMAIL is unset (org-service-key sessions), the
629	# strict rule accepts a bot-only commit with no trailer, so core.hooksPath
630	# is left untouched and none of this applies.
631	if [[ -n "$CCR_SESSION_ACCOUNT_EMAIL" ]]; then
632	  hooks_dir="$HOME/.ccr-git-hooks"
633	  mkdir -p "$hooks_dir" 2>/dev/null
634	
635	  # Passthrough stub for every standard client-side hook name except
636	  # commit-msg and prepare-commit-msg, which we own below. Each stub
637	  # execs the repo-local hook when present. Hooks live under the COMMON
638	  # git dir (the main repo's hooks/), not the per-worktree GIT_DIR, so
639	  # the lookup goes via rev-parse so linked worktrees resolve correctly.
640	  passthrough_hooks=(
641	    applypatch-msg pre-applypatch post-applypatch
642	    pre-commit pre-merge-commit post-commit
643	    pre-rebase post-checkout post-merge pre-push
644	    reference-transaction fsmonitor-watchman
645	    pre-auto-gc post-rewrite sendemail-validate post-index-change
646	  )
647	  for h in "${passthrough_hooks[@]}"; do
648	    cat >"$hooks_dir/$h" <<'PASSTHROUGH'
649	#!/bin/sh
650	common_dir=$(git rev-parse --git-common-dir 2>/dev/null) || common_dir=.git
651	local_hook="$common_dir/hooks/$(basename "$0")"
652	[ -x "$local_hook" ] && exec "$local_hook" "$@"
653	exit 0
